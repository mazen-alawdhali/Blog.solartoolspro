const CACHE_NAME = 'solar-blog-v4'; const ASSETS_TO_CACHE = [ './site.webmanifest', './favicon-96x96.png', './favicon.ico', './favicon.svg', './apple-touch-icon.png' ]; // تثبيت الخدمة وتخزين الأصول الثابتة فقط self.addEventListener('install', (event) => { event.waitUntil( caches.open(CACHE_NAME).then((cache) => { return cache.addAll(ASSETS_TO_CACHE); }) ); self.skipWaiting(); }); // تفعيل الخدمة وتنظيف الكاش القديم فوراً self.addEventListener('activate', (event) => { event.waitUntil( caches.keys().then((keys) => { return Promise.all( keys.map((key) => { if (key !== CACHE_NAME) { return caches.delete(key); } }) ); }) ); self.clients.claim(); }); // إدارة طلبات الشبكة (استراتيجية ذكية: جلب صفحات HTML و JSON من السيرفر مباشرة أولاً) self.addEventListener('fetch', (event) => { const url = new URL(event.request.url); // إذا كان الطلب لصفحة رئيسية، ملف HTML، أو articles.json -> اذهب للشبكة أولاً if (event.request.mode === 'navigate' || url.pathname.endsWith('.html') || url.pathname.includes('articles.json')) { event.respondWith( fetch(event.request) .then((networkResponse) => { return networkResponse; }) .catch(() => { return caches.match(event.request); }) ); } else { // للأصول الثابتة (صور، أيقونات) -> استخدم الكاش event.respondWith( caches.match(event.request).then((cachedResponse) => { return cachedResponse || fetch(event.request); }) ); } });
